cmake_minimum_required(VERSION 3.25)
project(vcf_external_sdk_check LANGUAGES C CXX)
find_package(OnistoneVCF 1.1 CONFIG REQUIRED)
# Repeated package lookup must not redefine or duplicate its imported targets.
find_package(OnistoneVCF 1.2 CONFIG REQUIRED)
if(NOT OnistoneVCF_ABI_VERSION STREQUAL "1.2")
 message(FATAL_ERROR "Unexpected SDK ABI version")
endif()
add_executable(sdk_external_c abi.c)
target_link_libraries(sdk_external_c PRIVATE OnistoneVCF::abi)
add_executable(sdk_external_cpp consumer.cpp)
target_link_libraries(sdk_external_cpp PRIVATE OnistoneVCF::sdk)
enable_testing()
add_test(NAME c_headers COMMAND sdk_external_c)
add_test(NAME cpp_discovery_without_provider COMMAND sdk_external_cpp)
