if(VCF_FUZZERS)

  message(FATAL_ERROR "Fuzzers require the Linux Clang sanitizer configuration")

 target_compile_options(vcf_core PRIVATE -fsanitize=fuzzer-no-link)
 execute_process(COMMAND git -C "${VCF_FUZZER_ROOT}" rev-parse HEAD OUTPUT_VARIABLE fuzzer_revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE fuzzer_git_status)
 if(NOT fuzzer_git_status EQUAL 0 OR NOT fuzzer_revision STREQUAL "87f0227cb60147a26a1eeb4fb06e3b505e9c7261")
  message(FATAL_ERROR "libFuzzer requires the locked LLVM 20.1.8 source commit")
 endif()
 execute_process(COMMAND git -C "${VCF_FUZZER_ROOT}" status --porcelain OUTPUT_VARIABLE fuzzer_changes OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE fuzzer_git_status)
 if(NOT fuzzer_git_status EQUAL 0 OR NOT fuzzer_changes STREQUAL "")
  message(FATAL_ERROR "libFuzzer source must be an unmodified pinned checkout")
 endif()
 file(GLOB fuzzer_sources CONFIGURE_DEPENDS "${VCF_FUZZER_ROOT}/compiler-rt/lib/fuzzer/Fuzzer*.cpp")
 add_library(vcf_fuzzer_runtime STATIC ${fuzzer_sources})
 # Distro libclang_rt.fuzzer was compiled against libstdc++. Compile the
 # official standalone sources with our libc++ flags, without instrumenting
 # libFuzzer itself. No second C++ standard library enters the process.
 target_compile_options(vcf_fuzzer_runtime PRIVATE -fno-sanitize=all)
