cmake_minimum_required(VERSION 3.25)
project(onistone_vcf VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(VCF_BUILD_PLUGIN "Build the Endstone native loader plugin" ON)
option(VCF_SANITIZERS "Address and undefined behavior sanitizers on Linux" OFF)
set(VCF_ENDSTONE_ROOT "" CACHE PATH "Pinned Endstone 0.11.10 checkout")
set(VCF_EXPECTED_ROOT "" CACHE PATH "Pinned expected-lite 0.9.0 checkout")
if(CMAKE_SIZEOF_VOID_P AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "VCF requires x86-64")
endif()
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
 add_compile_definitions(_ITERATOR_DEBUG_LEVEL=0 NOMINMAX WIN32_LEAN_AND_MEAN)
 add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/W4>" "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>" "$<$<COMPILE_LANGUAGE:CXX>:/permissive->" "$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>")
else()
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Linux builds require the pinned Clang/libc++ toolchain")
 endif()
 add_compile_options(-Wall -Wextra -Wpedantic -stdlib=libc++)
 add_link_options(-stdlib=libc++)
 if(VCF_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
 endif()
endif()
add_library(vcf_core STATIC framework/catalog.cpp framework/transactions.cpp framework/engine.cpp framework/abi.cpp framework/ledger.cpp)
target_include_directories(vcf_core PUBLIC include)
target_compile_definitions(vcf_core PRIVATE VCF_BUILD)
set_target_properties(vcf_core PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(VCF_BUILD_PLUGIN)
 include(cmake/EndstoneSdk.cmake)
 add_library(onistone_vcf SHARED framework/plugin.cpp)
 if(WIN32)
  enable_language(ASM_MASM)
  add_library(vcf_minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
  target_sources(onistone_vcf PRIVATE framework/platform/windows.cpp)
  target_sources(onistone_vcf PRIVATE framework/platform/windows/native_bridge.cpp framework/platform/windows/sign_distance_gate.asm framework/platform/windows/editor_distance_gate.asm)
  target_include_directories(onistone_vcf PRIVATE third_party third_party/minhook/include)
  target_link_libraries(onistone_vcf PRIVATE vcf_minhook)
  target_link_libraries(onistone_vcf PRIVATE bcrypt)
 else()
  target_sources(onistone_vcf PRIVATE framework/platform/linux.cpp)
 endif()
 target_link_libraries(onistone_vcf PRIVATE vcf_core endstone_sdk)
 target_compile_definitions(onistone_vcf PRIVATE VCF_BUILD)
 set_target_properties(onistone_vcf PROPERTIES PREFIX "endstone_" CXX_VISIBILITY_PRESET hidden)
 install(TARGETS onistone_vcf RUNTIME DESTINATION plugins LIBRARY DESTINATION plugins ARCHIVE DESTINATION sdk/lib)
 add_library(UICatalogShowcasePlugin SHARED examples/native/consumer.cpp)
 target_link_libraries(UICatalogShowcasePlugin PRIVATE endstone_sdk ${CMAKE_DL_LIBS})
 target_include_directories(UICatalogShowcasePlugin PRIVATE include)
 set_target_properties(UICatalogShowcasePlugin PROPERTIES PREFIX "endstone_" OUTPUT_NAME "vcf_catalog_showcase" CXX_VISIBILITY_PRESET hidden)
 install(TARGETS UICatalogShowcasePlugin RUNTIME DESTINATION examples LIBRARY DESTINATION examples)
 if(MSVC)
  target_link_options(onistone_vcf PRIVATE /DEBUG)
  install(FILES $<TARGET_PDB_FILE:onistone_vcf> DESTINATION symbols OPTIONAL)
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(vcf_core_tests tests/native/core_tests.cpp)
 target_link_libraries(vcf_core_tests PRIVATE vcf_core)
 add_test(NAME core COMMAND vcf_core_tests)
 add_executable(vcf_abi_c tests/native/abi_c.c)
 target_link_libraries(vcf_abi_c PRIVATE vcf_core)
 set_target_properties(vcf_abi_c PROPERTIES LINKER_LANGUAGE CXX)
 add_test(NAME c_abi_layout COMMAND vcf_abi_c)
endif()
add_executable(UICatalogShowcase examples/native/catalog_showcase.cpp)
target_link_libraries(UICatalogShowcase PRIVATE vcf_core)
install(FILES include/oni/vcf/abi.h include/oni/vcf/sdk.hpp include/oni/vcf/loader.hpp DESTINATION sdk/include/oni/vcf)
